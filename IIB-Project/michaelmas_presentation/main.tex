% ================= 1 MOTIVATION =================
\section{Motivation}

\begin{frame}{Why do short, slim nacelles care about crosswind?}
  \begin{columns}[T]
    \column{.52\textwidth}
    \begin{itemize}\setlength\itemsep{.45em}
      \item Low-FPR, large-fan engines need \textbf{shorter, thinner intakes} for lower drag and weight
      \item In strong crosswind the windward lip over-accelerates the flow $\rightarrow$ local supersonic region ended by a \textbf{shock}
      \item Shock--boundary-layer interaction (SBLI) can \alert{separate} the boundary layer $\rightarrow$ fan-plane distortion
      \item Thin lips = high curvature = worst case
    \end{itemize}
    \column{.45\textwidth}\centering
    \placeholder{5.6cm}{4.6cm}{Fig.: crosswind schematic\\(thesis Fig.\ 1.1 / 2.15)}
  \end{columns}
\end{frame}

\begin{frame}{Aims and research questions}
  \begin{columns}[T]
    \column{.48\textwidth}
    \card{teal}{Aim}{Characterise and quantify the unsteadiness of the shock and separation on the crosswind intake lip.}\\[6pt]
    \card{amber}{Comparison}{Baseline vs.\ \textbf{thinner} vs.\ \textbf{thicker} lip profile.}
    \column{.48\textwidth}
    \textbf{Research questions}
    \begin{enumerate}\setlength\itemsep{.4em}
      \item Where in the $(M_{\mathrm{entry}},\,\mathrm{MFS})$ space does unsteadiness occur?
      \item What are its frequency, amplitude and structure?
      \item How do these change with lip thickness?
      \item \textcolor{amber}{[Add / refine]}
    \end{enumerate}
  \end{columns}
\end{frame}

% ================= 2 BACKGROUND =================
\section{Background}

\begin{frame}{Transonic SBLI on a curved lip \hfill \chip{BACKGROUND}}
  \begin{columns}[T]
    \column{.5\textwidth}
    \begin{itemize}\setlength\itemsep{.4em}
      \item Flat-plate separation onset at shock Mach number $\approx 1.30$--$1.35$, roughly independent of shape factor $H$
      \item Interaction length $L^*/\delta^* = 70\,(H_i-1)$
      \item Curvature effects on onset: \textbf{disputed} in literature
      \item Separation bubbles can reduce shock drag but raise buffet risk
    \end{itemize}
    \column{.47\textwidth}\centering
    \placeholder{5.6cm}{4.4cm}{Fig.: attached vs.\ separated\\transonic SBLI}
  \end{columns}
\end{frame}

\begin{frame}{The CUED crosswind rig \hfill \chip{BACKGROUND}}
  \begin{columns}[T]
    \column{.55\textwidth}\centering
    \placeholder{6.6cm}{4.4cm}{Fig.: rig schematic with\\red / green / blue flow paths}
    \column{.42\textwidth}
    \begin{itemize}\setlength\itemsep{.4em}
      \item Quasi-2D, blow-down tunnel, full optical access
      \item $Re \approx 1\times10^{6}$, near-engine
      \item \textbf{Mass flow split} (MFS) $\sim$ engine power
      \item \textbf{Entry Mach} $M_{\mathrm{entry}}$ $\sim$ crosswind and power
      \item Matched 25 kt crosswind at $M_{\mathrm{entry}}=0.155$, MFS $=0.75$
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{What the previous work found \hfill \chip{BACKGROUND}}
  \begin{columns}[T]
    \column{.31\textwidth}\card{teal}{Separation onset}{Shock-induced separation only at a peak $M_{is}=1.44$ -- above flat-plate limit.}
    \column{.31\textwidth}\card{amber}{Unsteadiness}{Bubble forms, sheds, reattaches at $\sim$600\,Hz near critical $M_{\mathrm{entry}}\approx0.1575$.}
    \column{.31\textwidth}\card{coral}{Downstream effect}{LDV at the throat: intermittent reverse flow near the wall.}
  \end{columns}
  \vspace{.4cm}
  \begin{block}{The gap}
    Frequency was \emph{estimated by eye}, at one condition, on one lip. No quantification, no map, no geometry dependence.
  \end{block}
\end{frame}
